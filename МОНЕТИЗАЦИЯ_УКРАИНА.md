## 📋 Инструкция: Подключение LiqPay

### Шаг 1: Регистрация в LiqPay

1. Зайдите на https://www.liqpay.ua
2. Нажмите **"Регистрация"** или **"Войти"**
3. Заполните форму регистрации
4. Подтвердите email
5. Заполните данные компании/ИП
6. После верификации получите:
   - **public_key** (публичный ключ)
   - **private_key** (приватный ключ)

---

### Шаг 2: Установка библиотеки

В терминале выполните:

```bash
npm install liqpay-sdk-nodejs
```

### Шаг 3: Настройка на бэкенде

1. Откройте `server/index.js`
2. Добавьте в начало файла:

```javascript
const LiqPay = require('liqpay-sdk-nodejs')

const liqpay = new LiqPay(
  process.env.LIQPAY_PUBLIC_KEY,
  process.env.LIQPAY_PRIVATE_KEY
)
```

3. Добавьте API endpoints:

```javascript
// Создание платежа через LiqPay
app.post('/api/payments/create', async (req, res) => {
  try {
    const { amount, currency, description, consultationId } = req.body
    
    // Конвертируем валюту в UAH если нужно
    let amountInUAH = amount
    if (currency !== 'UAH') {
      // Здесь можно добавить конвертацию через API курсов валют
      // Для примера используем фиксированный курс
      const exchangeRates = {
        USD: 37, // примерный курс
        EUR: 40,
        RUB: 0.4
      }
      amountInUAH = amount * (exchangeRates[currency] || 1)
    }
    
    const orderId = `consultation_${consultationId}_${Date.now()}`
    
    const paymentParams = {
      version: '3',
      public_key: process.env.LIQPAY_PUBLIC_KEY,
      action: 'pay',
      amount: amountInUAH.toFixed(2),
      currency: 'UAH',
      description: description || `Оплата консультации #${consultationId}`,
      order_id: orderId,
      result_url: `${process.env.FRONTEND_URL}/payment/success`,
      server_url: `${process.env.API_URL || process.env.FRONTEND_URL}/api/payments/callback`,
      language: 'uk',
      sandbox: process.env.NODE_ENV === 'development' ? 1 : 0
    }
    
    // Создаем форму для оплаты
    const form = liqpay.cnb_form(paymentParams)
    
    res.json({
      orderId: orderId,
      formHtml: form,
      paymentUrl: `https://www.liqpay.ua/api/3/checkout?data=${liqpay.cnb_object(paymentParams).data}&signature=${liqpay.cnb_object(paymentParams).signature}`
    })
  } catch (error) {
    console.error('Ошибка создания платежа:', error)
    res.status(500).json({ error: 'Ошибка создания платежа' })
  }
})

// Callback от LiqPay (webhook)
app.post('/api/payments/callback', express.raw({ type: 'application/json' }), async (req, res) => {
  try {
    const data = req.body.data
    const signature = req.body.signature
    
    // Проверяем подпись
    const expectedSignature = liqpay.str_to_sign(process.env.LIQPAY_PRIVATE_KEY + data + process.env.LIQPAY_PRIVATE_KEY)
    
    if (signature !== expectedSignature) {
      return res.status(400).send('Invalid signature')
    }
    
    const decoded = Buffer.from(data, 'base64').toString('utf-8')
    const paymentData = JSON.parse(decoded)
    
    if (paymentData.status === 'success' || paymentData.status === 'sandbox') {
      const consultationId = paymentData.order_id.split('_')[1]
      // Обновить статус бронирования
      console.log(`Платеж успешен для консультации ${consultationId}`)
      // Здесь можно добавить логику обновления статуса в базе данных
    }
    
    res.status(200).send('OK')
  } catch (error) {
    console.error('Ошибка обработки callback:', error)
    res.status(500).send('Error')
  }
})

// Проверка статуса платежа
app.get('/api/payments/status/:orderId', async (req, res) => {
  try {
    const result = await liqpay.api('request', {
      action: 'status',
      version: '3',
      order_id: req.params.orderId
    })
    
    res.json({
      status: result.status,
      orderId: result.order_id,
      amount: result.amount,
      currency: result.currency
    })
  } catch (error) {
    console.error('Ошибка проверки статуса:', error)
    res.status(500).json({ error: 'Ошибка проверки статуса' })
  }
})
```

### Шаг 5: Обновите фронтенд

Откройте `src/contexts/PaymentContext.jsx` и замените функцию `processPayment`:

```javascript
const processPayment = async (paymentData) => {
  try {
    const API_BASE_URL = import.meta.env.VITE_API_URL || ''
    
    // Создаем платеж через бэкенд
    const response = await fetch(`${API_BASE_URL}/api/payments/create`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        amount: paymentData.amount,
        currency: paymentData.currency === 'UAH' ? 'UAH' : 'USD', // LiqPay работает с UAH
        description: `Оплата консультации ${paymentData.consultationId}`,
        consultationId: paymentData.consultationId
      })
    })
    
    if (!response.ok) {
      throw new Error('Ошибка создания платежа')
    }
    
    const payment = await response.json()
    
    // Перенаправляем на страницу оплаты LiqPay
    window.location.href = payment.paymentUrl
    
    return {
      success: true,
      orderId: payment.orderId
    }
  } catch (error) {
    console.error('Ошибка платежа:', error)
    return {
      success: false,
      error: error.message
    }
  }
}
```
